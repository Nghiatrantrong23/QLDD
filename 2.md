// ── 6. TẢI DỮ LIỆU NỀN ──────────────────────────────────────
async function loadThuaDat() {
  try {
    const res  = await fetch(GIS_CONFIG.apiGeoJson);
    const data = await res.json();
    if (state.thuaDatLayer) map.removeLayer(state.thuaDatLayer);
    state.thuaDatLayer = L.geoJSON(data, {
      style: f => ({
        color:       getLandColor(f.properties.loai_dat),
        weight:      1.5,
        fillOpacity: state.opacity['thua-dat'],
        fillColor:   getLandColor(f.properties.loai_dat),
      }),
      onEachFeature: (f, layer) => {
        const p = f.properties;
        layer.bindPopup(`
          <div style="font-family:'Be Vietnam Pro',sans-serif;min-width:200px">
            <div style="font-weight:700;font-size:13px;margin-bottom:8px">${p.ma_thua_dat||'N/A'}</div>
            <table style="font-size:12px;width:100%;border-collapse:collapse">
              <tr><td style="color:#666;padding:3px 0">Loại đất</td>  <td style="font-weight:600">${p.loai_dat||'—'}</td></tr>
              <tr><td style="color:#666;padding:3px 0">Diện tích</td> <td style="font-weight:600">${p.dien_tich?p.dien_tich.toFixed(1)+' m²':'—'}</td></tr>
              <tr><td style="color:#666;padding:3px 0">Địa chỉ</td>   <td>${p.dia_chi||'—'}</td></tr>
              <tr><td style="color:#666;padding:3px 0">Chủ SD</td>    <td>${p.chu_su_dung||'—'}</td></tr>
            </table>
            <div style="margin-top:10px;display:flex;gap:6px">
              <a href="/ho-so-dat/${p.id}/"     style="flex:1;padding:5px 8px;background:#2563eb;color:#fff;border-radius:5px;text-align:center;text-decoration:none;font-size:11px;font-weight:600">Chi tiết</a>
              <a href="/ho-so-dat/${p.id}/sua/" style="flex:1;padding:5px 8px;background:#f3f4f6;color:#374151;border-radius:5px;text-align:center;text-decoration:none;font-size:11px">Chỉnh sửa</a>
            </div>
          </div>
        `);
        layer.on('mouseover', () => layer.setStyle({ weight:2.5, fillOpacity: Math.min(state.opacity['thua-dat']+0.2, 1) }));

    showToast('Lỗi tải dữ liệu', 'Không thể tải lớp thửa đất', 'error');

// ── 7. CLICK BẢN ĐỒ → ĐẶT TÂM BUFFER ───────────────────────
map.on('click', function(e) {
  if (state.isDrawing || state.isMeasuring) return;
  if (state.activeTool !== 'buffer-click') return;

  const { lat, lng } = e.latlng;
  if (state.bufferMarker) map.removeLayer(state.bufferMarker);
  state.bufferMarker = L.circleMarker([lat, lng], {
    radius:6, color:'#2563eb', fillColor:'#2563eb', fillOpacity:1, weight:2,
  }).addTo(map).bindPopup(`Tâm buffer<br>${lat.toFixed(5)}, ${lng.toFixed(5)}`);

  setStatus('Đã chọn tâm buffer — Nhấn "Thực hiện Buffer" để phân tích', 'ready');
});

// ── 8. BUFFER TỪ ĐIỂM CLICK ─────────────────────────────────
async function runBufferClick() {
  if (!state.bufferMarker) {
    showToast('Chưa chọn tâm', 'Vui lòng click trên bản đồ để chọn điểm tâm', 'warning');
    return;
  }
  const { lat, lng } = state.bufferMarker.getLatLng();
  const radius    = parseInt(document.getElementById('radius-val').value)   || 300;
  const checkType = document.getElementById('check-type').value;
  const loaiDat   = document.getElementById('loai-dat-filter').value;

  showLoading('Đang tạo vùng đệm...', `Bán kính ${radius} m`);
  setStatus('Đang phân tích buffer...', 'busy');

  try {
    const res  = await apiFetch(GIS_CONFIG.apiBuffer, { lat, lng, ban_kinh:radius, kiem_tra:checkType, loai_dat:loaiDat });
    const data = await res.json();
    if (!res.ok) throw new Error(data.error || `HTTP ${res.status}`);
    displayBufferResult(data, radius);
    addHistoryEntry('Buffer điểm', `R=${radius}m, ${data.so_thua_trong_vung||0} thửa`, data.so_vi_pham > 0 ? 'vp' : 'ok');
  } catch(e) {
    showToast('Lỗi phân tích', e.message, 'error');
    setStatus('Lỗi — ' + e.message, 'error');
  } finally { hideLoading(); }
}

// ── 10. KIỂM TRA VI PHẠM ────────────────────────────────────
async function runIntersect() {

  const vungQH = document.getElementById('vung-qh-filter').value;

  // Nếu chọn "khu vực trên bản đồ" mà chưa vẽ vùng → cảnh báo

// ── 12. MẬT ĐỘ / HEATMAP ────────────────────────────────────
async function runDensity() {
  showLoading('Đang tạo bản đồ mật độ...', 'Tính kernel density');
  setStatus('Đang tạo heatmap...', 'busy');

  try {
    // Lấy trung tâm của tất cả thửa để vẽ heatmap đơn giản bằng circle markers
    const res  = await fetch(GIS_CONFIG.apiGeoJson);
    const data = await res.json();

    if (state.resultLayer) map.removeLayer(state.resultLayer);

    // Dùng circle marker với radius thay đổi theo mật độ (đơn giản)
    const points = [];
    (data.features || []).forEach(f => {
      if (f.geometry && f.geometry.type === 'Polygon') {
        const coords = f.geometry.coordinates[0];
        const cx = coords.reduce((s,c)=>s+c[0],0)/coords.length;
        const cy = coords.reduce((s,c)=>s+c[1],0)/coords.length;
        points.push([cy, cx]);
      } else if (f.geometry && f.geometry.type === 'Point') {
        points.push([f.geometry.coordinates[1], f.geometry.coordinates[0]]);
      }
    });

    state.resultLayer = L.layerGroup(
      points.map(p => L.circleMarker(p, {
        radius: 8, color:'transparent',
        fillColor:'#f43f5e', fillOpacity:0.15, weight:0,
      }))
    ).addTo(map);

    setStatus(`Heatmap — ${points.length} điểm`, 'ready');
    showToast('Bản đồ mật độ', `Hiển thị ${points.length} thửa đất`, 'success');
    addHistoryEntry('Mật độ thửa đất', `${points.length} điểm`, 'ok');
    renderDensityPanel(points.length);
    switchToResultsTab();
  } catch(e) {
    showToast('Lỗi', e.message, 'error');
    setStatus('Lỗi heatmap', 'error');
  } finally { hideLoading(); }
}

// ── 13. CLIP ─────────────────────────────────────────────────
async function runClip() {
  const vungSelect = document.querySelector('#params-clip .field-select');
  const vungId = vungSelect ? vungSelect.value : '';
  if (!vungId) { showToast('Chọn vùng', 'Vui lòng chọn vùng quy hoạch để clip', 'warning'); return; }

  showLoading('Đang clip dữ liệu...', 'Cắt thửa đất theo ranh giới vùng');
  setStatus('Đang clip...', 'busy');

  try {
    const res  = await apiFetch(GIS_CONFIG.apiIntersect, { pham_vi:'all', vung_qh_id:vungId, mode:'clip' });
    const data = await res.json();
    if (!res.ok) throw new Error(data.error || `HTTP ${res.status}`);

    if (state.resultLayer) map.removeLayer(state.resultLayer);
    if (data.result_geojson) {
      state.resultLayer = L.geoJSON(JSON.parse(data.result_geojson), {
        style: f => ({ color: getLandColor(f.properties.loai_dat), weight:2, fillOpacity:0.4, fillColor: getLandColor(f.properties.loai_dat) }),
      }).addTo(map);
      if (state.resultLayer.getBounds().isValid()) map.fitBounds(state.resultLayer.getBounds(), { padding:[30,30] });
    }

    setStatus(`Clip xong — ${data.so_thua||0} thửa trong vùng`, 'ready');
    showToast('Clip hoàn thành', `${data.so_thua||0} thửa trong vùng quy hoạch`, 'success');
    addHistoryEntry('Clip vùng QH', `${data.so_thua||0} thửa`, 'ok');
    renderClipPanel(data);
    switchToResultsTab();
  } catch(e) {
    showToast('Lỗi clip', e.message, 'error');
    setStatus('Lỗi clip', 'error');
  } finally { hideLoading(); }
}

// ── 14. HIỂN THỊ KẾT QUẢ BUFFER ────────────────────────────
function displayBufferResult(data, radius) {
  if (state.bufferLayer) map.removeLayer(state.bufferLayer);
  if (state.resultLayer) map.removeLayer(state.resultLayer);

  // Vẽ vòng buffer
  if (data.buffer_geojson) {
    state.bufferLayer = L.geoJSON(
      typeof data.buffer_geojson === 'string' ? JSON.parse(data.buffer_geojson) : data.buffer_geojson,
      { style:{ color:'#2563eb', weight:2, dashArray:'6,4', fillColor:'#2563eb', fillOpacity:0.06 } }
    ).addTo(map);
    map.fitBounds(state.bufferLayer.getBounds(), { padding:[40,40] });
  }

  // Vẽ thửa đất trong vùng (tô màu vi phạm)
  if (data.result_geojson) {
    const gj = typeof data.result_geojson === 'string' ? JSON.parse(data.result_geojson) : data.result_geojson;
    state.resultLayer = L.geoJSON(gj, {
      style: f => ({
        color:       f.properties.vi_pham ? '#f43f5e' : '#10b981',
        weight:      2.5,
        fillOpacity: 0.35,
        fillColor:   f.properties.vi_pham ? '#f43f5e' : '#10b981',
      }),
      onEachFeature: (f, layer) => {
        const p = f.properties;
        layer.bindPopup(`
          <b>${p.ma_thua_dat||'N/A'}</b><br>
          ${p.vi_pham ? '<span style="color:#f43f5e;font-weight:600">⚠ Vi phạm quy hoạch</span>' : '<span style="color:#10b981">✓ Hợp lệ</span>'}<br>
          Diện tích: ${p.dien_tich ? p.dien_tich.toFixed(1)+' m²' : '—'}
        `);
      },
    }).addTo(map);
  }

  // Result bubble
  document.getElementById('rb-title').textContent      = `Buffer ${radius}m`;
  document.getElementById('rb-so-thua').textContent     = `${data.so_thua_trong_vung||0} thửa`;
  document.getElementById('rb-vi-pham').textContent     = data.so_vi_pham > 0 ? `${data.so_vi_pham} thửa ⚠` : '0 thửa';
  document.getElementById('rb-dien-tich').textContent   = data.tong_dien_tich ? `${data.tong_dien_tich.toLocaleString()} m²` : '—';
  document.getElementById('rb-chu-su-dung').textContent = `${data.so_chu_su_dung||0} người`;
  document.getElementById('result-bubble').style.display = 'block';

  state.lastResults = { ...data, type:'buffer' };
  setStatus(`Hoàn thành — ${data.so_thua_trong_vung||0} thửa | ${data.so_vi_pham||0} vi phạm`, 'ready');

// ── 15. HIỂN THỊ KẾT QUẢ INTERSECT ─────────────────────────
function displayIntersectResult(data) {
  if (state.resultLayer) map.removeLayer(state.resultLayer);

  if (data.vi_pham_geojson) {
    const gj = typeof data.vi_pham_geojson === 'string' ? JSON.parse(data.vi_pham_geojson) : data.vi_pham_geojson;
    state.resultLayer = L.geoJSON(gj, {
      style: { color:'#f43f5e', weight:2.5, fillColor:'#f43f5e', fillOpacity:0.35 },

  document.getElementById('rb-title').textContent      = 'Vi phạm quy hoạch';

  setStatus(`Kiểm tra xong — ${data.so_vi_pham||0} thửa vi phạm`, 'ready');
  showToast('Kiểm tra vi phạm', `Phát hiện ${data.so_vi_pham||0} thửa vi phạm`, data.so_vi_pham > 0 ? 'warning' : 'success');

function zoomToParcel(id) {

      map.fitBounds(layer.getBounds ? layer.getBounds() : map.getBounds(), { padding:[30,30] });
      layer.openPopup();
    }
  });
}

async function loadHistoryItem(id) {
  if (!id) return;
  showLoading('Tải lại phân tích...', `#${id}`);
  try {
    const res  = await fetch(`${GIS_CONFIG.apiLichSu}${id}/`);
    const data = await res.json();
    if (!res.ok) throw new Error(data.error || 'Không tìm thấy');
    // Tùy theo loại phân tích mà xử lý
    if (data.loai === 'buffer') displayBufferResult(data.result, data.ban_kinh);
    else if (data.loai === 'intersect') displayIntersectResult(data.result);
    showToast('Đã tải lại', `Phân tích #${id}`, 'success');
  } catch(e) {
    showToast('Lỗi', e.message, 'error');
  } finally { hideLoading(); }
}

let measurePoints = [], measurePolyline = null;

document.getElementById('btn-measure').addEventListener('click', () => {
  const btn = document.getElementById('btn-measure');
  if (state.isMeasuring) {
    // Tắt đo
    state.isMeasuring = false;
    btn.classList.remove('active');
    map.off('click', measureClickHandler);
    measurePoints = [];
    if (measurePolyline) { map.removeLayer(measurePolyline); measurePolyline = null; }
    setStatus('Kết thúc đo lường', 'ready');
    map.getContainer().style.cursor = '';
  } else {
    state.isMeasuring = true;
    btn.classList.add('active');
    map.getContainer().style.cursor = 'crosshair';
    measurePoints = [];
    setStatus('Đo lường — Click để thêm điểm | Double-click để kết thúc', 'ready');
    map.on('click', measureClickHandler);
    map.once('dblclick', () => {
      state.isMeasuring = false;
      btn.classList.remove('active');
      map.off('click', measureClickHandler);
      map.getContainer().style.cursor = '';
      if (measurePoints.length >= 2) {
        const dist = calcDistance(measurePoints);
        showToast('Kết quả đo', `Khoảng cách: ${dist >= 1000 ? (dist/1000).toFixed(2)+' km' : dist.toFixed(0)+' m'}`, 'success');
      }
    });
  }
});

function measureClickHandler(e) {
  measurePoints.push(e.latlng);
  if (measurePolyline) map.removeLayer(measurePolyline);
  if (measurePoints.length >= 2) {
    measurePolyline = L.polyline(measurePoints, {
      color:'#f59e0b', weight:2, dashArray:'4,4',
    }).addTo(map);
    const dist = calcDistance(measurePoints);
    setStatus(`Khoảng cách tạm: ${dist >= 1000 ? (dist/1000).toFixed(2)+' km' : dist.toFixed(0)+' m'} — Double-click để kết thúc`, 'ready');
  }
  L.circleMarker(e.latlng, { radius:4, color:'#f59e0b', fillColor:'#f59e0b', fillOpacity:1, weight:1 }).addTo(map);
}

function calcDistance(points) {
  let total = 0;
  for (let i = 1; i < points.length; i++) total += points[i-1].distanceTo(points[i]);
  return total;
}
