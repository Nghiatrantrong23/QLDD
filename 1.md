map.on('click', function(e) {
  if (!state.clickMode || state.activeTool !== 'buffer-click') return;

  if (state.bufferMarker) map.removeLayer(state.bufferMarker);

async function runBufferClick() {
  if (!state.bufferMarker) {
    showToast('Chưa chọn tâm buffer', 'Vui lòng click trên bản đồ để chọn điểm tâm', 'warning');
    return;
  }
  const latlng = state.bufferMarker.getLatLng();
  const radius = parseInt(document.getElementById('radius-val').value) || 300;
  const checkType = document.getElementById('check-type').value;
  const loaiDat = document.getElementById('loai-dat-filter').value;

  showLoading('Đang tạo vùng đệm...', `Bán kính ${radius}m`);
  setStatus('Đang phân tích buffer...', 'busy');

  try {
    const res = await fetch(GIS_CONFIG.apiBuffer, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'X-CSRFToken': GIS_CONFIG.csrfToken },
      body: JSON.stringify({ lat: latlng.lat, lng: latlng.lng, ban_kinh: radius, kiem_tra: checkType, loai_dat: loaiDat }),
    });
    const data = await res.json();
    if (!res.ok) throw new Error(data.error || 'Lỗi server');
    displayBufferResult(data, radius);
  } catch(e) {
    showToast('Lỗi phân tích', e.message, 'error');
    setStatus('Lỗi — ' + e.message, 'error');
  } finally {
    hideLoading();
  }
}

function displayBufferResult(data, radius) {
  if (state.bufferLayer) map.removeLayer(state.bufferLayer);
  if (state.resultLayer) map.removeLayer(state.resultLayer);

  if (data.buffer_geojson) {
    state.bufferLayer = L.geoJSON(JSON.parse(data.buffer_geojson), {
      style: { color: '#2563eb', weight: 2, dashArray: '6,4', fillColor: '#2563eb', fillOpacity: 0.08 }
    }).addTo(map);
    map.fitBounds(state.bufferLayer.getBounds(), { padding: [40, 40] });
  }

  if (data.result_geojson) {
    state.resultLayer = L.geoJSON(JSON.parse(data.result_geojson), {
      style: (f) => ({
        color: f.properties.vi_pham ? '#f43f5e' : '#10b981',
        weight: 2.5, fillOpacity: 0.3,
        fillColor: f.properties.vi_pham ? '#f43f5e' : '#10b981',
      }),
    }).addTo(map);
  }

  // Result bubble
  document.getElementById('rb-title').textContent = `Kết quả Buffer ${radius}m`;
  document.getElementById('rb-so-thua').textContent = `${data.so_thua_trong_vung || 0} thửa`;
  document.getElementById('rb-vi-pham').textContent = data.so_vi_pham > 0 ? `${data.so_vi_pham} thửa ⚠` : `0 thửa`;
  document.getElementById('rb-dien-tich').textContent = data.tong_dien_tich ? `${data.tong_dien_tich.toLocaleString()} m²` : '—';
  document.getElementById('rb-chu-su-dung').textContent = `${data.so_chu_su_dung || 0} người`;
  document.getElementById('result-bubble').style.display = 'block';

  state.lastResults = data;
  setStatus(`Hoàn thành — ${data.so_thua_trong_vung || 0} thửa, ${data.so_vi_pham || 0} vi phạm`, 'ready');
  showToast('Buffer hoàn thành', `Tìm thấy ${data.so_thua_trong_vung || 0} thửa đất`, data.so_vi_pham > 0 ? 'warning' : 'success');
  renderResultsPanel(data);
}

function displayIntersectResult(data) {
  if (state.resultLayer) map.removeLayer(state.resultLayer);
  if (data.vi_pham_geojson) {
    state.resultLayer = L.geoJSON(JSON.parse(data.vi_pham_geojson), {
      style: { color: '#f43f5e', weight: 2.5, fillColor: '#f43f5e', fillOpacity: 0.35 },
      onEachFeature: (f, layer) => {
        layer.bindPopup(`<b>${f.properties.ma_thua_dat}</b><br>Vi phạm: ${f.properties.ten_vung_qh}`);
      }
    }).addTo(map);
    if (state.resultLayer.getBounds().isValid()) map.fitBounds(state.resultLayer.getBounds(), { padding: [40,40] });
  }
  setStatus(`Kiểm tra xong — ${data.so_vi_pham || 0} thửa vi phạm`, 'ready');
  showToast('Kiểm tra vi phạm', `Phát hiện ${data.so_vi_pham || 0} thửa vi phạm quy hoạch`, data.so_vi_pham > 0 ? 'warning' : 'success');
  renderResultsPanel(data);
}
